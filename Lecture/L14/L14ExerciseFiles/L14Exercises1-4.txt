/* Exercise 1:  Find sailors who reserved boat 102 on September 8, 2025.
				Include in the query result objects fields sailor id and name. */
// QUERY RESULT: {sailorId: 64, sName: 'Horatio'}
// <<< Construct your MQL query below this line >>>

db.sailors.find(
	{reservations: {$elemMatch: {boatId: 102, rDate: '08-SEP-25'}}},
	{_id: 0, sailorId: 1, sName: 1}
);

// ----------------------------------------------------------------------------
/* Exercise 2:  Find sailors who reserved boats 101 or 102 but not boats 103 or 
				104.
				Include in the query result objects fields sailor id and name. */
// QUERY RESULT: {sailorId: 64, sName: 'Horatio'} {sailorId: 99, sName: 'Chris'}
// <<< Construct your MQL query below this line >>>

db.sailors.find(
	{$and: [
		{'reservations.boatId' : {$in:[101, 102]}},
		{'reservations.boatId' : {$nin:[103, 104]}},
	]},
	{_id: 0, sailorId: 1, sName: 1}
);

// ----------------------------------------------------------------------------
/* Exercise 3:	Find reservations made for either boat 101 or boat 103.
				Include in the query result objects fields sailor id, name and 
				reservations. */
// QUERY RESULT:
// {sailorId: 22, sName: 'Dustin',  reservations: [{boatId: 101, rDate: '10-OCT-25'},
//												   {boatId: 103, rDate: '08-SEP-25'}]}
// {sailorId: 31, sName: 'Lucy',    reservations: [{boatId: 103, rDate: '06-NOV-25'}]}
// {sailorId: 64, sName: 'Horatio', reservations: [{boatId: 101, rDate: '05-SEP-25'}]}
// {sailorId: 74, sName: 'Horatio', reservations: [{boatId: 103, rDate: '08-SEP-25'}]}
// <<< Construct your MQL query below this line >>>

db.sailors.find(
	{'reservations.boatId' : {$in:[101, 103]}},
	{_id: 0, sailorId: 1, sName: 1, 
	reservations: {$filter: 
					{
						input: '$reservations',
						as: 'ress',
						cond: { $in: ["$$ress.boatId", [101, 103]] }
					}
	}}
);
